def sha(path):

def encoded_q(value):

def primary_row(primary, source='positive_B', setting='coarse', k=64, order=24, dps='80'):

    control = next(c for c in record['controls'] if c['quadrature_order'] == order)
    return next(r for r in control['levels'][dps] if r['K'] == k)


def independent_row(independent, source='positive_B', setting='coarse', k=64):
